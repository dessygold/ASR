apiVersion: argoproj.io/v1alpha1
kind: Application
metadata:
  name: nexusiq
  namespace: argocd
  annotations:
    argocd.argoproj.io/sync-wave: "2"
  finalizers:
    - resources-finalizer.argocd.argoproj.io
  labels:
    system: {{ .Values.system }}
    component: application
spec:
  project: {{ .Values.system }}
  sources:
    - repoURL: 186542266927.dkr.ecr.us-gov-west-1.amazonaws.com # core-test
      chart: charts/nexus-iq-server-ha
      targetRevision: 209.0.0
      helm:
        valueFiles:
          - $values/helm-values/values.yaml
          - $values/helm-values/values-{{ .Values.environment }}.yaml
    - repoURL: https://code.cdso.army.mil/dsop/apps/nexusiq-ha.git
      targetRevision: main
      ref: values
    # support manifests
    - repoURL: https://code.cdso.army.mil/dsop/apps/nexusiq-ha.git
      targetRevision: main
      path: support-manifests
  destination:
    name: {{ .Values.system }}
    namespace: nexusiq
  syncPolicy:
    automated:
      prune: true
      selfHeal: true
    syncOptions:
    - CreateNamespace=true
    # avoid syncing db migration job more than once
    - ApplyOutOfSyncOnly=true

{{- end }}
